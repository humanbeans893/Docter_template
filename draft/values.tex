% Numbers used in the text. Each research keeps its own in projects/<id>/values.tex, registered
% below so that every value is defined before \begin{document} (the abstracts come first).
% Define: \deftextvalue{<id>-<name>}{<value as written>} or \defvalue{<id>-<name>}{<plain number>}.
% Use: \val{<id>-<name>} (a \defvalue number is typeset with siunitx \num),
% \valqty{<id>-<name>}{<unit>} for a quantity, e.g. \valqty{<id>-DepthMax}{\metre}
% (a \deftextvalue value prints as value\,\unit{unit}), and \rawval{...} for the bare string.
% A missing value inside \qty{\rawval{...}}{...} stops even draft builds (siunitx: Invalid
% number); \valqty marks it instead.
\newcommand{\defvalue}[2]{\expandafter\gdef\csname thesisvalue@#1\endcsname{#2}}
\newcommand{\deftextvalue}[2]{\defvalue{#1}{#2}\expandafter\gdef\csname thesisvaluetext@#1\endcsname{}}
\newcommand{\rawval}[1]{\ifcsname thesisvalue@#1\endcsname\csname thesisvalue@#1\endcsname\else\ThesisMissingValue{#1}\fi}
\newcommand{\valqty}[2]{%
  \ifcsname thesisvalue@#1\endcsname
    \ifcsname thesisvaluetext@#1\endcsname
      \csname thesisvalue@#1\endcsname\,\unit{#2}%
    \else
      \qty{\csname thesisvalue@#1\endcsname}{#2}%
    \fi
  \else
    \ThesisMissingValue{#1}%
  \fi}
\newcommand{\val}[1]{%
  \ifcsname thesisvalue@#1\endcsname
    \ifcsname thesisvaluetext@#1\endcsname
      \csname thesisvalue@#1\endcsname
    \else
      \num{\csname thesisvalue@#1\endcsname}%
    \fi
  \else
    \ThesisMissingValue{#1}%
  \fi}
% \ThesisMissingValue and \ThesisValues: final build -> error; draft and review -> warning + marker.
\ifThesisFinal
  \newcommand{\ThesisMissingValue}[1]{\PackageError{thesis}{Undefined value `#1'}{Define it in projects/<id>/values.tex and register that file in draft/values.tex.}}
  \newcommand{\ThesisValues}[1]{\InputIfFileExists{#1}{}{\PackageError{thesis}{Values file `#1' not found}{Copy projects/_template/values.tex to it.}}}
\else
  % Robust: in a heading or caption the marker reaches the .toc/.lof unexpanded. Expanded, a key
  % with `_' (ids may contain it) is read back as a subscript and breaks the next run.
  \DeclareRobustCommand{\ThesisMissingValue}[1]{\PackageWarning{thesis}{Undefined value `#1'}\textbf{??}\texttt{\detokenize{#1}}}
  \newcommand{\ThesisValues}[1]{\InputIfFileExists{#1}{}{\PackageWarning{thesis}{Values file `#1' not found}}}
\fi
% One line per research in research.json order, path relative to draft/ without .tex.
