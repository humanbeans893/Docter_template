% --- Format ------------------------------------------------------------------------------------
% Headings in the serif bold of the text (ltjsreport defaults to sans-serif gothic), chapter
% titles single-spaced, and the English names and footnote marks (1, not *1).
\renewcommand{\headfont}{\bfseries}
\makeatletter
\patchcmd{\@makechapterhead}{\normalfont}{\normalfont\setstretch{1}}{}{}
\patchcmd{\@makeschapterhead}{\normalfont}{\normalfont\setstretch{1}}{}{}
\makeatother

\renewcommand{\appendixname}{Appendix~}
